    ┌─────────────────────────────────────────────────────────────────┐
    │                      INPUT HANDLER                              │
    │                                                                 │
    │  Input:  image_path                                            │
    │  Action: Load X-ray image from disk                            │
    │  Output: image_data (bytes), image_name                        │
    └──────────────────────────┬──────────────────────────────────────┘
                               │
                               ▼
    ┌─────────────────────────────────────────────────────────────────┐
    │                   MULTIMODAL PROCESSOR                          │
    │                                                                 │
    │  Input:  image_data                                            │
    │  Action: Send to LM Studio with expert prompt                  │
    │          "You are an expert radiologist..."                    │
    │  Output: {                                                     │
    │            "Pneumonia": "Present" or "Absent",                 │
    │            "Atelectasis": "Present" or "Absent",               │
    │            "Fracture": "Present" or "Absent"                   │
    │          }                                                     │
    └──────────────────────────┬──────────────────────────────────────┘
                               │
                               ▼
    ┌─────────────────────────────────────────────────────────────────┐
    │                     RESULT EVALUATOR                            │
    │                                                                 │
    │  Input:  prediction, image_path                                │
    │  Action: Look up ground truth from CSV                         │
    │          Compare each condition                                │
    │  Output: ground_truth = {                                      │
    │            "Pneumonia": 1.0 or 0.0,                            │
    │            "Atelectasis": 1.0 or 0.0,                          │
    │            "Fracture": 1.0 or 0.0                              │
    │          }                                                     │
    │          evaluation = {                                        │
    │            "Pneumonia": True/False,                            │
    │            "Atelectasis": True/False,                          │
    │            "Fracture": True/False                              │
    │          }                                                     │
    └──────────────────────────┬──────────────────────────────────────┘
                               │
                               ▼
    ┌─────────────────────────────────────────────────────────────────┐
    │                      OUTPUT HANDLER                             │
    │                                                                 │
    │  Input:  Complete state                                        │
    │  Action: Pass through (for future extensions)                  │
    │  Output: Final state                                           │
    └──────────────────────────┬──────────────────────────────────────┘
